"""Validador determinista del PM (Etapa 6, Art. 4.4).

Rechaza-e-informa: NO modifica la propuesta del PM. Si viola un límite duro,
devuelve los errores concretos y el PM re-propone (bucle acotado aguas arriba).
La frontera queda *enforced*, no confiada a la buena conducta del LLM.

Límites que enforcea (Art. 4.1/4.3):
  - solo empresas ELEGIBLES (nada inventado fuera del conjunto);
  - peso de cada posición en [piso, hard cap];
  - máx posiciones por sector;
  - cap por factor macro (conteo completo en multi-pertenencia, Art. 4.5);
  - el invertido no supera el máximo que deja el cash objetivo del operador (Art. 4.7).
"""
from __future__ import annotations

from dataclasses import dataclass, field

from backend.nucleo.constantes import MAX_POR_SUBSECTOR, PISO_POSICION

# Máx % de cartera expuesto a un mismo factor macro (calibrable — registro 14).
FACTOR_CAP = 0.40


@dataclass
class Candidata:
    """Lo que el validador sabe (determinista) de cada nombre elegible."""
    ticker: str
    elegible: bool
    hard_cap: float
    sector: str | None = None
    factores: list[str] = field(default_factory=list)   # factores macro (conteo completo)


@dataclass
class LineaPropuesta:
    """Una línea de la propuesta del PM."""
    ticker: str
    peso: float


def validar_propuesta(lineas: list[LineaPropuesta], candidatas: list[Candidata], *,
                      cash_objetivo: float, piso: float = PISO_POSICION,
                      max_por_sector: int = MAX_POR_SUBSECTOR,
                      factor_cap: float = FACTOR_CAP, tol: float = 1e-9) -> dict:
    """Valida la propuesta del PM contra los límites duros. Devuelve
    {ok, errores, invertido, cash}. No modifica nada."""
    por_ticker = {c.ticker.upper(): c for c in candidatas}
    errores: list[str] = []
    por_sector: dict[str, int] = {}
    por_factor: dict[str, float] = {}
    invertido = 0.0
    vistos: set[str] = set()

    for l in lineas:
        t = l.ticker.upper()
        c = por_ticker.get(t)
        if c is None:
            errores.append(f"{t}: fuera del conjunto elegible (ticker inventado)")
            continue
        if t in vistos:
            errores.append(f"{t}: duplicado en la propuesta")
            continue
        vistos.add(t)

        if not c.elegible:
            errores.append(f"{t}: no elegible")
        if l.peso < piso - tol:
            errores.append(f"{t}: peso {l.peso:.1%} < piso {piso:.0%}")
        if l.peso > c.hard_cap + tol:
            errores.append(f"{t}: peso {l.peso:.1%} > hard cap {c.hard_cap:.0%}")

        invertido += l.peso
        if c.sector:
            por_sector[c.sector] = por_sector.get(c.sector, 0) + 1
        for f in c.factores:
            por_factor[f] = por_factor.get(f, 0.0) + l.peso

    for s, n in por_sector.items():
        if n > max_por_sector:
            errores.append(f"sector {s}: {n} posiciones > máx {max_por_sector}")
    for f, w in por_factor.items():
        if w > factor_cap + tol:
            errores.append(f"factor {f}: {w:.1%} > cap {factor_cap:.0%}")

    maximo_invertible = 1.0 - cash_objetivo
    if invertido > maximo_invertible + tol:
        errores.append(
            f"invertido {invertido:.1%} > máximo {maximo_invertible:.0%} "
            f"(cash objetivo {cash_objetivo:.0%})")

    return {"ok": not errores, "errores": errores,
            "invertido": invertido, "cash": 1.0 - invertido}

"""Deploy de un mix aprobado al ledger fiel (Etapa 6).

Convierte pesos objetivo en **fills reales** de acciones enteras sobre el ledger
(Art. 11): ventas antes que compras (§7.4), fill a un precio dado (el caller lo
setea a apertura+30′), con slippage + comisión, respetando el **cap de participación
sobre ADV** y sin dejar la caja en negativo. Determinista; testeable con datos
inyectados.
"""
from __future__ import annotations

from datetime import date

from sqlalchemy.orm import Session

from backend.ledger.core import proyectar, registrar_fill


def desplegar(session: Session, *, mix: dict, precios: dict, fecha: date,
              adv_usd: dict | None = None, participacion_adv: float = 0.15,
              slippage_bps: float = 25.0, comision_por_accion: float = 0.005,
              comision_min: float = 1.0) -> dict:
    """Lleva la cartera hacia `mix` (dict ticker->peso). Un ticker con peso 0 o
    ausente se vende a cero. Devuelve {ventas, compras, cash, invertido, equity}.
    """
    precios = {k.upper(): v for k, v in precios.items()}
    adv_usd = {k.upper(): v for k, v in (adv_usd or {}).items()}
    mix_peso = {k.upper(): v for k, v in mix.items()}

    est = proyectar(session, precios)
    equity = est["equity"]
    pos = est["posiciones"]

    # ---- Calcular órdenes (delta a target, en acciones enteras) ----
    ordenes: list[tuple[str, int, float]] = []
    for t in set(mix_peso) | set(pos):
        precio = precios.get(t)
        if not precio or precio <= 0:
            continue  # sin precio válido no se opera
        target_val = mix_peso.get(t, 0.0) * equity
        cur_shares = pos[t].shares if t in pos else 0.0
        delta_sh = round((target_val - cur_shares * precio) / precio)
        if delta_sh == 0:
            continue
        # Cap de participación sobre ADV.
        adv = adv_usd.get(t)
        if adv and adv > 0:
            max_sh = int(participacion_adv * adv / precio)
            if abs(delta_sh) > max_sh:
                delta_sh = max_sh if delta_sh > 0 else -max_sh
        if delta_sh != 0:
            ordenes.append((t, delta_sh, precio))

    ordenes.sort(key=lambda o: o[1])   # negativas (ventas) primero
    ventas, compras = [], []

    def _fill(t, n, precio, lado):
        eff = precio * (1 + slippage_bps / 10_000) if lado == "buy" else precio * (1 - slippage_bps / 10_000)
        comision = max(comision_min, comision_por_accion * n)
        registrar_fill(session, ticker=t, lado=lado, shares=n, precio=eff,
                       comision=comision, slippage_bps=slippage_bps, fecha_fill=fecha)
        (compras if lado == "buy" else ventas).append({"ticker": t, "shares": n, "precio": eff})

    for t, sh, precio in ordenes:
        if sh < 0:
            _fill(t, abs(sh), precio, "sell")

    for t, sh, precio in ordenes:
        if sh <= 0:
            continue
        n = sh
        # Guard de caja: no dejar la caja en negativo (compra parcial si hace falta).
        cash = proyectar(session, precios)["caja"]
        eff = precio * (1 + slippage_bps / 10_000)

        def _costo(k: int) -> float:
            return k * eff + max(comision_min, comision_por_accion * k)

        if _costo(n) > cash:
            n = int(cash / (eff + comision_por_accion))   # estimación con comisión por acción
            while n > 0 and _costo(n) > cash:
                n -= 1
        if n > 0:
            _fill(t, n, precio, "buy")

    final = proyectar(session, precios)
    return {"ventas": ventas, "compras": compras, "cash": final["caja"],
            "invertido": final["valor_posiciones"], "equity": final["equity"]}

"""Tests del validador determinista del PM (Etapa 6, Art. 4.4)."""
import pytest

from backend.seleccion.validador import (
    Candidata,
    LineaPropuesta,
    validar_propuesta,
)


def _cands():
    return [
        Candidata("ACME", elegible=True, hard_cap=0.25, sector="ciber", factores=["capex hyperscaler"]),
        Candidata("BETA", elegible=True, hard_cap=0.20, sector="ciber", factores=["capex hyperscaler"]),
        Candidata("GAMA", elegible=True, hard_cap=0.30, sector="energia", factores=["defensa"]),
        Candidata("DELT", elegible=False, hard_cap=0.20, sector="fotonica"),
    ]


def _val(lineas, cash=0.0, **kw):
    return validar_propuesta([LineaPropuesta(t, p) for t, p in lineas], _cands(),
                             cash_objetivo=cash, **kw)


def test_propuesta_valida():
    r = _val([("ACME", 0.20), ("GAMA", 0.25)], cash=0.55)
    assert r["ok"], r["errores"]
    assert r["invertido"] == pytest.approx(0.45)


def test_ticker_inventado():
    r = _val([("ZZZZ", 0.10)])
    assert not r["ok"]
    assert any("inventado" in e for e in r["errores"])


def test_no_elegible():
    r = _val([("DELT", 0.10)], cash=0.9)
    assert not r["ok"]
    assert any("no elegible" in e for e in r["errores"])


def test_supera_hard_cap():
    r = _val([("BETA", 0.25)], cash=0.75)   # BETA cap 0.20
    assert not r["ok"]
    assert any("hard cap" in e for e in r["errores"])


def test_debajo_del_piso():
    r = _val([("ACME", 0.02)], cash=0.9)    # piso 5%
    assert not r["ok"]
    assert any("piso" in e for e in r["errores"])


def test_cap_por_sector():
    # 3 en "ciber" con max 2 → error
    r = _val([("ACME", 0.10), ("BETA", 0.10)], cash=0.8, max_por_sector=1)
    assert not r["ok"]
    assert any("sector ciber" in e for e in r["errores"])


def test_cap_por_factor():
    # ACME 0.25 + BETA 0.20 = 0.45 en "capex hyperscaler" > 0.40
    r = _val([("ACME", 0.25), ("BETA", 0.20)], cash=0.55)
    assert not r["ok"]
    assert any("factor capex hyperscaler" in e for e in r["errores"])


def test_sobre_despliegue_vs_cash():
    # invertido 0.60 pero cash objetivo 0.50 → máximo invertible 0.50
    r = _val([("ACME", 0.25), ("GAMA", 0.30), ("BETA", 0.05)], cash=0.50)
    assert not r["ok"]
    assert any("invertido" in e for e in r["errores"])


def test_duplicado():
    r = _val([("ACME", 0.10), ("ACME", 0.10)], cash=0.8)
    assert not r["ok"]
    assert any("duplicado" in e for e in r["errores"])

"""Tests del deploy al ledger (Etapa 6). SQLite in-memory."""
from datetime import date

import pytest
from sqlalchemy import create_engine
from sqlalchemy.orm import Session

from backend.ledger import core
from backend.ledger.models import crear_tablas
from backend.seleccion.deploy import desplegar

CAPITAL = 100_000.0


@pytest.fixture
def sesion():
    engine = create_engine("sqlite:///:memory:", future=True)
    crear_tablas(engine)
    with Session(engine, future=True) as s:
        core.depositar(s, CAPITAL, date(2026, 1, 1))
        yield s


def test_deploy_desde_cash_compra(sesion):
    r = desplegar(sesion, mix={"ACME": 0.5}, precios={"ACME": 50}, fecha=date(2026, 1, 2),
                  slippage_bps=0.0, comision_por_accion=0.0, comision_min=0.0)
    assert len(r["compras"]) == 1 and r["compras"][0]["shares"] == 1000
    assert r["invertido"] == pytest.approx(50_000)
    assert r["cash"] == pytest.approx(50_000)


def test_reconciliacion_tras_deploy(sesion):
    desplegar(sesion, mix={"ACME": 0.4, "BETA": 0.3}, precios={"ACME": 50, "BETA": 20},
              fecha=date(2026, 1, 2))
    rec = core.reconciliar(sesion, {"ACME": 50, "BETA": 20})
    assert rec["ok"], rec["violaciones"]


def test_deploy_vende_lo_que_sale_del_mix(sesion):
    desplegar(sesion, mix={"ACME": 0.5}, precios={"ACME": 50}, fecha=date(2026, 1, 2),
              slippage_bps=0.0, comision_por_accion=0.0, comision_min=0.0)
    # Nuevo mix sin ACME → se vende a cero.
    r = desplegar(sesion, mix={"BETA": 0.3}, precios={"ACME": 50, "BETA": 20},
                  fecha=date(2026, 1, 3), slippage_bps=0.0, comision_por_accion=0.0, comision_min=0.0)
    est = core.proyectar(sesion, {"ACME": 50, "BETA": 20})
    assert "ACME" not in est["posiciones"]
    assert "BETA" in est["posiciones"]
    assert any(v["ticker"] == "ACME" for v in r["ventas"])


def test_cap_adv_limita_la_orden(sesion):
    # ADV $100k, participación 15% → máx $15k/día = 300 acciones @50 (vs 1000 objetivo).
    r = desplegar(sesion, mix={"ACME": 0.5}, precios={"ACME": 50}, fecha=date(2026, 1, 2),
                  adv_usd={"ACME": 100_000}, slippage_bps=0.0, comision_por_accion=0.0, comision_min=0.0)
    assert r["compras"][0]["shares"] == 300


def test_no_deja_caja_negativa(sesion):
    # Pide 100% en ACME con costos → el guard evita caja negativa.
    r = desplegar(sesion, mix={"ACME": 1.0}, precios={"ACME": 50}, fecha=date(2026, 1, 2),
                  slippage_bps=50.0, comision_por_accion=0.01, comision_min=1.0)
    assert r["cash"] >= 0


def test_ventas_antes_que_compras(sesion):
    # Cartera con ACME; nuevo mix cambia a BETA → debe vender ACME (libera cash) y comprar BETA.
    desplegar(sesion, mix={"ACME": 0.9}, precios={"ACME": 50}, fecha=date(2026, 1, 2),
              slippage_bps=0.0, comision_por_accion=0.0, comision_min=0.0)
    r = desplegar(sesion, mix={"BETA": 0.9}, precios={"ACME": 50, "BETA": 20},
                  fecha=date(2026, 1, 3), slippage_bps=0.0, comision_por_accion=0.0, comision_min=0.0)
    assert r["ventas"] and r["compras"]
    assert r["cash"] >= 0
