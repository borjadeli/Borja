"""Tablas del ledger — nx_* (Etapa 2, init_docs/16).

Base propia (LedgerBase) → aislada de backend/models.py: `crear_tablas` solo crea
las nx_*, no toca las tablas del sistema viejo. Todo el log es APPEND-ONLY;
posición y caja se derivan en backend/ledger/core.py.
"""
from __future__ import annotations

from datetime import date, datetime

from sqlalchemy import (
    Boolean,
    Date,
    DateTime,
    Float,
    Integer,
    String,
    Text,
    UniqueConstraint,
)
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column


class LedgerBase(DeclarativeBase):
    pass


class NxCashMovimiento(LedgerBase):
    """Movimientos de caja que NO son fills: depósito inicial, tasa libre de
    riesgo, ajustes. (Dividendos y cierres entran vía NxCorporateAction.)"""
    __tablename__ = "nx_cash_movimiento"
    id: Mapped[int] = mapped_column(Integer, primary_key=True, autoincrement=True)
    fecha: Mapped[date] = mapped_column(Date, index=True)
    tipo: Mapped[str] = mapped_column(String(32))       # deposito_inicial | rf | ajuste
    monto: Mapped[float] = mapped_column(Float)          # signed
    nota: Mapped[str | None] = mapped_column(Text, nullable=True)


class NxFill(LedgerBase):
    """Ejecución real (Art. 11). Append-only. cash_delta = efecto neto en caja."""
    __tablename__ = "nx_fill"
    id: Mapped[int] = mapped_column(Integer, primary_key=True, autoincrement=True)
    ticker: Mapped[str] = mapped_column(String(16), index=True)
    lado: Mapped[str] = mapped_column(String(4))         # buy | sell
    shares: Mapped[float] = mapped_column(Float)         # > 0
    precio: Mapped[float] = mapped_column(Float)         # fill = apertura + 30' (Art. 11.1)
    comision: Mapped[float] = mapped_column(Float, default=0.0)
    slippage_bps: Mapped[float] = mapped_column(Float, default=0.0)
    cash_delta: Mapped[float] = mapped_column(Float)
    fecha_fill: Mapped[date] = mapped_column(Date, index=True)
    fecha_decision: Mapped[datetime | None] = mapped_column(DateTime, nullable=True)
    fecha_dato: Mapped[datetime | None] = mapped_column(DateTime, nullable=True)
    input_hash: Mapped[str | None] = mapped_column(String(64), nullable=True)
    nota: Mapped[str | None] = mapped_column(Text, nullable=True)


class NxCorporateAction(LedgerBase):
    """Split / dividendo / cierre (M&A o delisting). Append-only (Art. 11.6)."""
    __tablename__ = "nx_corporate_action"
    id: Mapped[int] = mapped_column(Integer, primary_key=True, autoincrement=True)
    ticker: Mapped[str] = mapped_column(String(16), index=True)
    tipo: Mapped[str] = mapped_column(String(16))        # split | dividendo | cierre
    fecha: Mapped[date] = mapped_column(Date, index=True)
    ratio: Mapped[float | None] = mapped_column(Float, nullable=True)             # split
    monto_por_accion: Mapped[float | None] = mapped_column(Float, nullable=True)  # dividendo
    valor_recupero: Mapped[float | None] = mapped_column(Float, nullable=True)    # cierre
    nota: Mapped[str | None] = mapped_column(Text, nullable=True)


class NxSnapshot(LedgerBase):
    """Marca diaria (Art. 11.5): equity, caja, doble P&L, benchmark. Append-only."""
    __tablename__ = "nx_snapshot"
    id: Mapped[int] = mapped_column(Integer, primary_key=True, autoincrement=True)
    fecha: Mapped[date] = mapped_column(Date, index=True)
    equity: Mapped[float] = mapped_column(Float)
    caja: Mapped[float] = mapped_column(Float)
    valor_posiciones: Mapped[float] = mapped_column(Float)
    pnl_ideal: Mapped[float | None] = mapped_column(Float, nullable=True)
    pnl_modelado: Mapped[float | None] = mapped_column(Float, nullable=True)
    benchmark: Mapped[float | None] = mapped_column(Float, nullable=True)
    n_posiciones: Mapped[int] = mapped_column(Integer, default=0)

    LedgerBase.metadata.create_all(bind=engine)

"""Lógica del ledger: append al log + proyección determinista a posición/caja.

Funciones puras sobre una Session inyectada (testeable con SQLite in-memory). El
estado actual (posiciones, caja, equity) es SIEMPRE una proyección del log
inmutable — nunca se guarda mutado. Ver init_docs/16_ledger_etapa2.md.
"""
from __future__ import annotations

from dataclasses import dataclass
from datetime import date, datetime

from sqlalchemy import select
from sqlalchemy.orm import Session

from backend.ledger.models import (
    NxCashMovimiento,
    NxCorporateAction,
    NxFill,
    NxSnapshot,
)


@dataclass
class Posicion:
    ticker: str
    shares: float = 0.0
    costo_total: float = 0.0      # base total (incluye comisiones de compra)
    realized_pnl: float = 0.0

    @property
    def costo_promedio(self) -> float:
        return self.costo_total / self.shares if self.shares else 0.0


# ---------------------------- Escritura (append-only) ----------------------------

def depositar(session: Session, monto: float, fecha: date,
              tipo: str = "deposito_inicial", nota: str | None = None) -> NxCashMovimiento:
    mov = NxCashMovimiento(fecha=fecha, tipo=tipo, monto=monto, nota=nota)
    session.add(mov)
    session.flush()
    return mov


def registrar_fill(session: Session, *, ticker: str, lado: str, shares: float,
                   precio: float, comision: float = 0.0, slippage_bps: float = 0.0,
                   fecha_fill: date, fecha_decision: datetime | None = None,
                   fecha_dato: datetime | None = None, input_hash: str | None = None,
                   nota: str | None = None) -> NxFill:
    ticker = ticker.upper()
    if lado not in ("buy", "sell"):
        raise ValueError(f"lado inválido: {lado!r}")
    if shares <= 0 or precio <= 0:
        raise ValueError("shares y precio deben ser > 0")
    bruto = shares * precio
    cash_delta = -(bruto + comision) if lado == "buy" else (bruto - comision)
    fill = NxFill(ticker=ticker, lado=lado, shares=shares, precio=precio,
                  comision=comision, slippage_bps=slippage_bps, cash_delta=cash_delta,
                  fecha_fill=fecha_fill, fecha_decision=fecha_decision,
                  fecha_dato=fecha_dato, input_hash=input_hash, nota=nota)
    session.add(fill)
    session.flush()
    return fill


def aplicar_corporate_action(session: Session, *, ticker: str, tipo: str, fecha: date,
                             ratio: float | None = None, monto_por_accion: float | None = None,
                             valor_recupero: float | None = None,
                             nota: str | None = None) -> NxCorporateAction:
    ticker = ticker.upper()
    if tipo not in ("split", "dividendo", "cierre"):
        raise ValueError(f"corporate action inválida: {tipo!r}")
    ca = NxCorporateAction(ticker=ticker, tipo=tipo, fecha=fecha, ratio=ratio,
                           monto_por_accion=monto_por_accion,
                           valor_recupero=valor_recupero, nota=nota)
    session.add(ca)
    session.flush()
    return ca


# ---------------------------- Proyección (log → estado) ----------------------------

def _eventos_ordenados(session: Session) -> list[tuple]:
    """Log completo en orden cronológico determinista. Misma fecha: primero
    movimientos de caja, luego corporate actions, luego fills."""
    eventos: list[tuple] = []
    for m in session.execute(select(NxCashMovimiento)).scalars():
        eventos.append((m.fecha, 0, m.id, "cash", m))
    for c in session.execute(select(NxCorporateAction)).scalars():
        eventos.append((c.fecha, 1, c.id, "ca", c))
    for f in session.execute(select(NxFill)).scalars():
        eventos.append((f.fecha_fill, 2, f.id, "fill", f))
    eventos.sort(key=lambda e: (e[0], e[1], e[2]))
    return eventos


def proyectar(session: Session, precios: dict | None = None) -> dict:
    """Reconstruye caja + posiciones desde el log. Determinista y puro.

    precios: dict ticker->precio para valuar (sin precio de un ticker → valor 0).
    Devuelve {caja, posiciones, valor_posiciones, equity, realized_pnl}.
    """
    precios = {k.upper(): v for k, v in (precios or {}).items()}
    caja = 0.0
    pos: dict[str, Posicion] = {}

    for _fecha, _rank, _id, kind, ev in _eventos_ordenados(session):
        if kind == "cash":
            caja += ev.monto
        elif kind == "fill":
            p = pos.setdefault(ev.ticker, Posicion(ev.ticker))
            caja += ev.cash_delta
            if ev.lado == "buy":
                p.shares += ev.shares
                p.costo_total += ev.shares * ev.precio + ev.comision
            else:  # sell
                avg = p.costo_promedio
                p.realized_pnl += ev.shares * ev.precio - ev.comision - ev.shares * avg
                p.shares -= ev.shares
                p.costo_total -= ev.shares * avg
                if p.shares <= 1e-9:
                    p.shares = 0.0
                    p.costo_total = 0.0
        elif kind == "ca":
            p = pos.get(ev.ticker)
            if not p or p.shares <= 0:
                continue
            if ev.tipo == "split" and ev.ratio:
                p.shares *= ev.ratio          # costo_total intacto → avg /= ratio
            elif ev.tipo == "dividendo" and ev.monto_por_accion:
                caja += p.shares * ev.monto_por_accion
            elif ev.tipo == "cierre":
                recupero = ev.valor_recupero or 0.0
                avg = p.costo_promedio
                p.realized_pnl += p.shares * recupero - p.shares * avg
                caja += p.shares * recupero
                p.shares = 0.0
                p.costo_total = 0.0

    posiciones = {t: pp for t, pp in pos.items() if pp.shares > 0}
    valor_posiciones = sum(pp.shares * precios.get(t, 0.0) for t, pp in posiciones.items())
    equity = caja + valor_posiciones
    return {
        "caja": caja,
        "posiciones": posiciones,
        "valor_posiciones": valor_posiciones,
        "equity": equity,
        "realized_pnl": sum(pp.realized_pnl for pp in pos.values()),
    }


def snapshot(session: Session, *, fecha: date, precios: dict,
             benchmark: float | None = None, capital_inicial: float | None = None,
             pnl_ideal: float | None = None) -> NxSnapshot:
    est = proyectar(session, precios)
    pnl_modelado = (est["equity"] - capital_inicial) if capital_inicial is not None else None
    snap = NxSnapshot(fecha=fecha, equity=est["equity"], caja=est["caja"],
                      valor_posiciones=est["valor_posiciones"], pnl_ideal=pnl_ideal,
                      pnl_modelado=pnl_modelado, benchmark=benchmark,
                      n_posiciones=len(est["posiciones"]))
    session.add(snap)
    session.flush()
    return snap


def reconciliar(session: Session, precios: dict, *, tol: float = 1e-6) -> dict:
    """Chequea los invariantes de reconciliación (init_docs/16 §E). Devuelve
    {ok, violaciones, estado}."""
    est = proyectar(session, precios)
    precios = {k.upper(): v for k, v in precios.items()}
    violaciones: list[str] = []

    # Inv 1 — caja + Σ(shares×precio) == equity.
    recomputo = est["caja"] + sum(p.shares * precios.get(t, 0.0)
                                  for t, p in est["posiciones"].items())
    if abs(recomputo - est["equity"]) > tol:
        violaciones.append(f"caja+posiciones ({recomputo:.6f}) != equity ({est['equity']:.6f})")

    # Inv 7 — pesos + cash% == 100%.
    if est["equity"] > 0:
        suma = est["caja"] / est["equity"] + sum(
            p.shares * precios.get(t, 0.0) for t, p in est["posiciones"].items()) / est["equity"]
        if abs(suma - 1.0) > tol:
            violaciones.append(f"pesos+cash ({suma:.6f}) != 1.0")

    # Inv 3 — ninguna posición sin fill que la respalde.
    tickers_con_fill = {f.ticker for f in session.execute(select(NxFill)).scalars()}
    for t in est["posiciones"]:
        if t not in tickers_con_fill:
            violaciones.append(f"posición {t} sin fill que la respalde")

    return {"ok": not violaciones, "violaciones": violaciones, "estado": est}

"""Identidad, máquina de estados (Art. 5) y mapa persistido — sobre el ledger.

Reusa VIDAS_REENTRADA del núcleo (no redefine constantes). El estado actual de una
empresa es la proyección del log de eventos (nx_empresa_evento); `nx_empresa.estado`
es un cache que se actualiza en la misma transacción. Transiciones no permitidas se
rechazan (fallo seguro, Art. 8).
"""
from __future__ import annotations

from datetime import date, datetime
from enum import Enum

from sqlalchemy import select
from sqlalchemy.orm import Session

from backend.ledger.models import (
    NxEmpresa,
    NxEmpresaEvento,
    NxEmpresaNodo,
    NxNodo,
)
from backend.nucleo.constantes import VIDAS_REENTRADA


class Estado(str, Enum):
    CANDIDATA = "Candidata"
    DESCARTADA = "Descartada"
    ESPERANDO_EVIDENCIA = "Esperando-evidencia"
    EN_DEEP_RESEARCH = "En-deep-research"
    ELEGIBLE = "Elegible"
    WATCHLIST = "Watchlist"
    POSICION = "Posicion"
    TESIS_INVALIDADA = "Tesis-invalidada"
    ARCHIVADA = "Archivada"


# Transiciones permitidas (Art. 5.2). Claves/valores como strings.
TRANSICIONES: dict[str, set[str]] = {
    Estado.CANDIDATA.value: {Estado.DESCARTADA.value, Estado.ESPERANDO_EVIDENCIA.value,
                             Estado.EN_DEEP_RESEARCH.value},
    Estado.ESPERANDO_EVIDENCIA.value: {Estado.EN_DEEP_RESEARCH.value, Estado.DESCARTADA.value,
                                       Estado.ARCHIVADA.value},
    Estado.EN_DEEP_RESEARCH.value: {Estado.ELEGIBLE.value, Estado.DESCARTADA.value,
                                    Estado.ESPERANDO_EVIDENCIA.value},
    Estado.ELEGIBLE.value: {Estado.POSICION.value, Estado.WATCHLIST.value},
    Estado.WATCHLIST.value: {Estado.ELEGIBLE.value, Estado.POSICION.value,
                             Estado.ESPERANDO_EVIDENCIA.value, Estado.ARCHIVADA.value},
    Estado.POSICION.value: {Estado.TESIS_INVALIDADA.value, Estado.WATCHLIST.value},
    Estado.TESIS_INVALIDADA.value: {Estado.ARCHIVADA.value},
    Estado.DESCARTADA.value: {Estado.CANDIDATA.value},   # fría, nuevo hecho
    Estado.ARCHIVADA.value: {Estado.CANDIDATA.value},    # re-entrada limitada (Candado B)
}


def _v(e) -> str:
    return e.value if isinstance(e, Estado) else str(e)


def _dt(f) -> datetime:
    return f if isinstance(f, datetime) else datetime(f.year, f.month, f.day)


def _append_evento(session: Session, ticker: str, anterior: str | None, nuevo: str,
                   disparado_por: str, motivo: str | None, fecha: datetime,
                   ficha_version_id: int | None = None) -> NxEmpresaEvento:
    ev = NxEmpresaEvento(ticker=ticker, estado_anterior=anterior, estado_nuevo=nuevo,
                         disparado_por=disparado_por, motivo=motivo, fecha=fecha,
                         ficha_version_id=ficha_version_id)
    session.add(ev)
    session.flush()
    return ev


def crear_candidata(session: Session, *, ticker: str, fecha: date, nombre: str | None = None,
                    sector: str | None = None, jurisdiccion: str | None = None,
                    exchange: str | None = None, fuente_ingreso: str = "manual",
                    nodo_id: int | None = None, motivo: str | None = None) -> NxEmpresa:
    """Ingresa (o re-ingresa) un ticker como Candidata. Si ya existe, es una
    transición a Candidata (con Candado B si venía de Archivada)."""
    ticker = ticker.upper()
    emp = session.get(NxEmpresa, ticker)
    if emp is None:
        emp = NxEmpresa(ticker=ticker, nombre=nombre, sector=sector, jurisdiccion=jurisdiccion,
                        exchange=exchange, estado=Estado.CANDIDATA.value,
                        fuente_ingreso=fuente_ingreso, fecha_ingreso=fecha, reentradas_usadas=0)
        session.add(emp)
        session.flush()
        _append_evento(session, ticker, None, Estado.CANDIDATA.value, "sistema",
                       motivo or "ingreso", _dt(fecha))
    else:
        transicionar(session, ticker=ticker, estado_nuevo=Estado.CANDIDATA,
                     disparado_por="sistema", motivo=motivo or "re-ingreso", fecha=fecha)
    if nodo_id is not None:
        vincular_nodo(session, ticker=ticker, nodo_id=nodo_id, es_primario=True)
    return emp


def transicionar(session: Session, *, ticker: str, estado_nuevo, disparado_por: str,
                 fecha: date, motivo: str | None = None,
                 ficha_version_id: int | None = None) -> NxEmpresa:
    """Cambia el estado de una empresa validando el Art. 5.2. Append al log +
    actualiza el cache. Rechaza transiciones no permitidas y el 2º re-ingreso
    (Candado B)."""
    ticker = ticker.upper()
    emp = session.get(NxEmpresa, ticker)
    if emp is None:
        raise ValueError(f"empresa inexistente: {ticker}")
    actual = emp.estado
    destino = _v(estado_nuevo)
    if destino not in TRANSICIONES.get(actual, set()):
        raise ValueError(f"transición no permitida: {actual} → {destino}")
    # Candado B: re-entrada Archivada → Candidata, máx VIDAS_REENTRADA.
    if actual == Estado.ARCHIVADA.value and destino == Estado.CANDIDATA.value:
        if emp.reentradas_usadas >= VIDAS_REENTRADA:
            raise ValueError(
                f"Candado B: {ticker} ya usó su re-entrada (máx {VIDAS_REENTRADA})")
        emp.reentradas_usadas += 1
    _append_evento(session, ticker, actual, destino, disparado_por, motivo, _dt(fecha),
                   ficha_version_id)
    emp.estado = destino
    session.flush()
    return emp


def estado_desde_log(session: Session, ticker: str) -> str | None:
    """Estado actual reconstruido desde el log (para reconciliar contra el cache)."""
    ev = session.execute(
        select(NxEmpresaEvento).where(NxEmpresaEvento.ticker == ticker.upper())
        .order_by(NxEmpresaEvento.id.desc())
    ).scalars().first()
    return ev.estado_nuevo if ev else None


# ---------------------------- Mapa ----------------------------

def crear_nodo(session: Session, *, nombre: str, cluster: str | None = None,
               factor_macro: str | None = None, estado_confirmacion: str = "promesa",
               nota: str | None = None) -> NxNodo:
    n = NxNodo(nombre=nombre, cluster=cluster, factor_macro=factor_macro,
               estado_confirmacion=estado_confirmacion, nota=nota)
    session.add(n)
    session.flush()
    return n


def vincular_nodo(session: Session, *, ticker: str, nodo_id: int,
                  es_primario: bool = False) -> NxEmpresaNodo:
    """Vincula empresa↔nodo. Idempotente; NUNCA borra (Art. 5.3)."""
    ticker = ticker.upper()
    existe = session.execute(
        select(NxEmpresaNodo).where(NxEmpresaNodo.ticker == ticker,
                                    NxEmpresaNodo.nodo_id == nodo_id)
    ).scalar_one_or_none()
    if existe:
        return existe
    v = NxEmpresaNodo(ticker=ticker, nodo_id=nodo_id, es_primario=es_primario)
    session.add(v)
    session.flush()
    return v

def _hasta_archivada(sesion, ticker, base_day):
    """Lleva un ticker vivo hasta Archivada por la vía de tesis rota."""
    e.transicionar(sesion, ticker=ticker, estado_nuevo=Estado.EN_DEEP_RESEARCH,
                   disparado_por="regla", fecha=date(2026, 1, base_day))
    e.transicionar(sesion, ticker=ticker, estado_nuevo=Estado.ELEGIBLE,
                   disparado_por="regla", fecha=date(2026, 1, base_day + 1))
    e.transicionar(sesion, ticker=ticker, estado_nuevo=Estado.POSICION,
                   disparado_por="pm", fecha=date(2026, 1, base_day + 2))
    e.transicionar(sesion, ticker=ticker, estado_nuevo=Estado.TESIS_INVALIDADA,
                   disparado_por="regla", fecha=date(2026, 1, base_day + 3))
    e.transicionar(sesion, ticker=ticker, estado_nuevo=Estado.ARCHIVADA,
                   disparado_por="regla", fecha=date(2026, 1, base_day + 4))
