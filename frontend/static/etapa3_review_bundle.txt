########## FILE: backend/ingreso/universo.py ##########

"""Filtro de universo (Art. 2) — ¿esta empresa es ADMISIBLE para analizarse?

Determinista y puro: hechos inyectados → admisible/no + motivos. NO decide si es
buena inversión (eso es el gate) ni si vale el deep research (eso es el Filtro 0);
solo si PUEDE entrar al pipeline. Piso de market cap SIN techo (decisión del
operador). Reusa LIQUIDEZ_SCREENER del núcleo.
"""
from __future__ import annotations

from dataclasses import dataclass, field

from backend.nucleo.constantes import LIQUIDEZ_SCREENER

# Piso de market cap (sin techo — el filtrado por tamaño lo hace el pipeline, Art. 2.3).
MIN_MARKET_CAP = 50_000_000

# Jurisdicciones permitidas (Art. 2.1): EE.UU., Canadá, Europa, Asia tech. China afuera.
JURISDICCIONES_PERMITIDAS = frozenset({
    "US", "CA",                       # Norteamérica
    "EU", "UK", "DE", "FR", "NL", "SE", "CH",  # Europa
    "JP", "KR", "TW",                 # Asia tech relevante
})


@dataclass
class HechosUniverso:
    jurisdiccion: str | None = None
    es_adr: bool = False
    market_cap: float | None = None
    adv_usd: float | None = None      # volumen promedio diario en USD
    tiene_nodo: bool = False          # vínculo con un nodo del mapa (Art. 2.1)


@dataclass
class ResultadoUniverso:
    admisible: bool
    motivos: list[str] = field(default_factory=list)


def evaluar_universo(h: HechosUniverso) -> ResultadoUniverso:
    """¿Admisible para el pipeline? Todos los criterios deben cumplirse."""
    motivos: list[str] = []
    if h.jurisdiccion not in JURISDICCIONES_PERMITIDAS:
        motivos.append(f"jurisdicción no permitida: {h.jurisdiccion or 'N/D'}")
    if h.es_adr:
        motivos.append("ADR (no se opera vía ADR)")
    if h.market_cap is None or h.market_cap < MIN_MARKET_CAP:
        cap = f"${h.market_cap/1e6:.0f}M" if h.market_cap is not None else "N/D"
        motivos.append(f"market cap {cap} < ${MIN_MARKET_CAP/1e6:.0f}M")
    if h.adv_usd is None or h.adv_usd < LIQUIDEZ_SCREENER:
        liq = f"${h.adv_usd/1e6:.1f}M" if h.adv_usd is not None else "N/D"
        motivos.append(f"liquidez {liq}/día < ${LIQUIDEZ_SCREENER/1e6:.1f}M/día")
    if not h.tiene_nodo:
        motivos.append("sin vínculo con nodo del mapa")
    return ResultadoUniverso(admisible=not motivos, motivos=motivos)

########## FILE: backend/ingreso/filtro0.py ##########

"""Filtro 0 (Art. 6.5) — ¿vale la pena el deep research (caro)?

Determinista y SIN LLM (el pre-screen busca recall; un LLM mete varianza invisible).
Hechos inyectados → uno de 3 veredictos: 'descartar' | 'esperar-evidencia' |
'investigar'. Cada veredicto lleva el motivo (para taguear la descartada y el
re-chequeo dirigido). Reforma barata del screener de 5 filtros ya existente.
"""
from __future__ import annotations

from dataclasses import dataclass

# Umbrales (calibrables — registro 14). Espíritu del screener viejo + constitución.
CREC_OK = 0.20          # crecimiento YoY "bueno"
CREC_NEARMISS = 0.10    # 10-20% → casi
MARGEN_OK = 0.30        # margen bruto "bueno"
MARGEN_NEARMISS = 0.20  # 20-30% → casi
DEUDA_ALTA = 2.0        # deuda/revenue
RUNWAY_MIN_MESES = 6    # financiación mínima antes de considerar riesgo crítico

VEREDICTOS = ("descartar", "esperar-evidencia", "investigar")


@dataclass
class HechosFiltro0:
    es_prerevenue: bool = False
    tiene_contrato_firme: bool = False       # pre-revenue: contrato/LOI económica firme
    tiene_traccion_comercial: bool = False   # señal comercial no-firme (design-in, pipeline)
    crecimiento_yoy: float | None = None
    margen_bruto: float | None = None
    deuda_sobre_revenue: float | None = None
    runway_meses: float | None = None


@dataclass
class ResultadoFiltro0:
    veredicto: str
    motivo: str


def _nivel(valor: float | None, ok: float, near: float) -> str:
    """ok | near | feo. Valor faltante → feo (default conservador)."""
    if valor is None:
        return "feo"
    if valor >= ok:
        return "ok"
    if valor >= near:
        return "near"
    return "feo"


def evaluar_filtro0(h: HechosFiltro0) -> ResultadoFiltro0:
    """Veredicto barato de 'vale el deep research'. Determinista."""
    # 1. Riesgo financiero duro → descartar (el deep research solo confirmaría la muerte).
    if (h.runway_meses is not None and h.runway_meses < RUNWAY_MIN_MESES
            and (h.deuda_sobre_revenue or 0.0) > DEUDA_ALTA):
        return ResultadoFiltro0(
            "descartar",
            f"riesgo financiero: runway {h.runway_meses:.0f}m (<{RUNWAY_MIN_MESES}m) "
            f"+ deuda/rev {h.deuda_sobre_revenue:.1f}x",
        )

    # 2. Pre-revenue: exige evidencia comercial firme (Art. 2.1).
    if h.es_prerevenue:
        if h.tiene_contrato_firme:
            return ResultadoFiltro0("investigar", "pre-revenue con contrato firme")
        if h.tiene_traccion_comercial:
            return ResultadoFiltro0(
                "esperar-evidencia",
                "pre-revenue sin contrato firme, con tracción → esperar el hecho")
        return ResultadoFiltro0("descartar", "pre-revenue sin evidencia comercial")

    # 3. Con revenue: calidad mínima (crecimiento + margen).
    nc = _nivel(h.crecimiento_yoy, CREC_OK, CREC_NEARMISS)
    nm = _nivel(h.margen_bruto, MARGEN_OK, MARGEN_NEARMISS)
    niveles = {nc, nm}
    crec_s = f"{h.crecimiento_yoy:.0%}" if h.crecimiento_yoy is not None else "N/D"
    margen_s = f"{h.margen_bruto:.0%}" if h.margen_bruto is not None else "N/D"

    if niveles == {"ok"}:
        return ResultadoFiltro0("investigar", f"crecimiento {crec_s} y margen {margen_s} ok")
    if "feo" in niveles:
        return ResultadoFiltro0(
            "descartar", f"calidad insuficiente (crecimiento {crec_s}, margen {margen_s})")
    # Solo 'near' (ninguno feo) → casi; esperar el próximo print.
    return ResultadoFiltro0(
        "esperar-evidencia", f"cerca en crecimiento/margen ({crec_s}/{margen_s}) → esperar print")

########## FILE: backend/ingreso/flujo.py ##########

"""Orquestación de la Etapa 3 sobre el ledger.

Entrada (manual o discovery) → filtro de universo → crear Candidata → Filtro 0 →
transición de estado. Más el hook de re-emisión por cambio de NODO (Art. 5.3): las
descartadas vinculadas a un nodo que cambió se re-chequean barato (Filtro 0, sin LLM).
"""
from __future__ import annotations

from datetime import date

from sqlalchemy import select
from sqlalchemy.orm import Session

from backend.ingreso.filtro0 import HechosFiltro0, evaluar_filtro0
from backend.ingreso.universo import HechosUniverso, evaluar_universo
from backend.ledger.entidades import Estado, crear_candidata, transicionar
from backend.ledger.models import NxEmpresa, NxEmpresaNodo

VEREDICTO_A_ESTADO = {
    "descartar": Estado.DESCARTADA,
    "esperar-evidencia": Estado.ESPERANDO_EVIDENCIA,
    "investigar": Estado.EN_DEEP_RESEARCH,
}


def ingresar(session: Session, *, ticker: str, universo: HechosUniverso,
             filtro0: HechosFiltro0, fecha: date, nodo_id: int | None = None,
             fuente_ingreso: str = "manual", nombre: str | None = None,
             sector: str | None = None) -> dict:
    """Corre una candidata por universo + Filtro 0 y la deja en el estado que toque.

    Devuelve {resultado, motivo?}. 'fuera_universo' → NO se crea Candidata.
    """
    u = evaluar_universo(universo)
    if not u.admisible:
        return {"resultado": "fuera_universo", "motivos": u.motivos}

    crear_candidata(session, ticker=ticker, fecha=fecha, nodo_id=nodo_id,
                    fuente_ingreso=fuente_ingreso, nombre=nombre, sector=sector,
                    jurisdiccion=universo.jurisdiccion)

    f = evaluar_filtro0(filtro0)
    transicionar(session, ticker=ticker, estado_nuevo=VEREDICTO_A_ESTADO[f.veredicto],
                 disparado_por="regla", motivo=f.motivo, fecha=fecha)
    return {"resultado": f.veredicto, "motivo": f.motivo}


def reemitir_por_nodo(session: Session, *, nodo_id: int,
                      filtro0_por_ticker: dict[str, HechosFiltro0], fecha: date) -> dict:
    """Hook Art. 5.3: un nodo cambió → re-chequea (Filtro 0 barato) las DESCARTADAS
    vinculadas. Si el veredicto mejora, revive Descartada → Candidata → destino.

    filtro0_por_ticker: hechos frescos por ticker. Los que no estén se saltean.
    Devuelve {ticker: veredicto_nuevo}.
    """
    resultados: dict[str, str] = {}
    vinculos = session.execute(
        select(NxEmpresaNodo).where(NxEmpresaNodo.nodo_id == nodo_id)
    ).scalars().all()

    for v in vinculos:
        emp = session.get(NxEmpresa, v.ticker)
        if emp is None or emp.estado != Estado.DESCARTADA.value:
            continue
        h = filtro0_por_ticker.get(v.ticker)
        if h is None:
            continue
        f = evaluar_filtro0(h)
        if f.veredicto == "descartar":
            resultados[v.ticker] = "sigue-descartada"
            continue
        # Revive: Descartada → Candidata (no gasta Candado B) → destino del veredicto.
        transicionar(session, ticker=v.ticker, estado_nuevo=Estado.CANDIDATA,
                     disparado_por="regla",
                     motivo=f"re-emisión por cambio de nodo {nodo_id}", fecha=fecha)
        transicionar(session, ticker=v.ticker, estado_nuevo=VEREDICTO_A_ESTADO[f.veredicto],
                     disparado_por="regla", motivo=f.motivo, fecha=fecha)
        resultados[v.ticker] = f.veredicto

    return resultados

"""Tests de la Etapa 3: filtro de universo, Filtro 0 y el flujo sobre el ledger.

SQLite in-memory — no toca data/borja.db.
"""
from datetime import date

import pytest
from sqlalchemy import create_engine
from sqlalchemy.orm import Session

from backend.ingreso import flujo
from backend.ingreso.filtro0 import HechosFiltro0, evaluar_filtro0
from backend.ingreso.universo import HechosUniverso, evaluar_universo
from backend.ledger import entidades as e
from backend.ledger.entidades import Estado
from backend.ledger.models import NxEmpresa, crear_tablas


# ------------------------------- Universo -------------------------------

def _universo_ok(**kw):
    base = dict(jurisdiccion="US", es_adr=False, market_cap=500e6, adv_usd=5e6,
                tiene_nodo=True)
    base.update(kw)
    return HechosUniverso(**base)


def test_universo_admisible():
    assert evaluar_universo(_universo_ok()).admisible


@pytest.mark.parametrize("kw,motivo", [
    (dict(jurisdiccion="CN"), "jurisdicción"),
    (dict(es_adr=True), "ADR"),
    (dict(market_cap=10e6), "market cap"),
    (dict(adv_usd=100e3), "liquidez"),
    (dict(tiene_nodo=False), "nodo"),
])
def test_universo_rechaza(kw, motivo):
    r = evaluar_universo(_universo_ok(**kw))
    assert not r.admisible
    assert any(motivo in m for m in r.motivos)


def test_universo_sin_techo_de_market_cap():
    # Una mega-cap NO se rechaza por tamaño (Art. 2.3).
    assert evaluar_universo(_universo_ok(market_cap=3_000_000e6)).admisible


# ------------------------------- Filtro 0 -------------------------------

def test_filtro0_riesgo_financiero_descarta():
    h = HechosFiltro0(runway_meses=3, deuda_sobre_revenue=3.0, crecimiento_yoy=0.5,
                      margen_bruto=0.6)
    assert evaluar_filtro0(h).veredicto == "descartar"


def test_filtro0_prerevenue():
    assert evaluar_filtro0(HechosFiltro0(es_prerevenue=True,
                                         tiene_contrato_firme=True)).veredicto == "investigar"
    assert evaluar_filtro0(HechosFiltro0(es_prerevenue=True,
                                         tiene_traccion_comercial=True)).veredicto == "esperar-evidencia"
    assert evaluar_filtro0(HechosFiltro0(es_prerevenue=True)).veredicto == "descartar"


def test_filtro0_revenue_investigar():
    assert evaluar_filtro0(HechosFiltro0(crecimiento_yoy=0.30,
                                         margen_bruto=0.45)).veredicto == "investigar"


def test_filtro0_revenue_near_miss_espera():
    assert evaluar_filtro0(HechosFiltro0(crecimiento_yoy=0.15,
                                         margen_bruto=0.35)).veredicto == "esperar-evidencia"
    assert evaluar_filtro0(HechosFiltro0(crecimiento_yoy=0.25,
                                         margen_bruto=0.25)).veredicto == "esperar-evidencia"


def test_filtro0_revenue_feo_descarta():
    assert evaluar_filtro0(HechosFiltro0(crecimiento_yoy=0.05,
                                         margen_bruto=0.10)).veredicto == "descartar"
    # una ok y otra feo → descarta (una dimensión demasiado débil)
    assert evaluar_filtro0(HechosFiltro0(crecimiento_yoy=0.40,
                                         margen_bruto=0.05)).veredicto == "descartar"


# ------------------------------- Flujo (sobre el ledger) -------------------------------

@pytest.fixture
def sesion():
    engine = create_engine("sqlite:///:memory:", future=True)
    crear_tablas(engine)
    with Session(engine, future=True) as s:
        yield s


def test_flujo_fuera_universo_no_crea_candidata(sesion):
    r = flujo.ingresar(sesion, ticker="ACME", fecha=date(2026, 1, 1),
                       universo=_universo_ok(jurisdiccion="CN"),
                       filtro0=HechosFiltro0(crecimiento_yoy=0.5, margen_bruto=0.6))
    assert r["resultado"] == "fuera_universo"
    assert sesion.get(NxEmpresa, "ACME") is None


def test_flujo_investigar_deja_en_deep_research(sesion):
    nodo = e.crear_nodo(sesion, nombre="HBM", cluster="datacenter")
    r = flujo.ingresar(sesion, ticker="ACME", fecha=date(2026, 1, 1), nodo_id=nodo.id,
                       universo=_universo_ok(), filtro0=HechosFiltro0(crecimiento_yoy=0.3,
                                                                      margen_bruto=0.45))
    assert r["resultado"] == "investigar"
    assert e.estado_desde_log(sesion, "ACME") == Estado.EN_DEEP_RESEARCH.value


def test_flujo_descartar_deja_descartada(sesion):
    nodo = e.crear_nodo(sesion, nombre="HBM", cluster="datacenter")
    r = flujo.ingresar(sesion, ticker="ACME", fecha=date(2026, 1, 1), nodo_id=nodo.id,
                       universo=_universo_ok(), filtro0=HechosFiltro0(crecimiento_yoy=0.02,
                                                                      margen_bruto=0.05))
    assert r["resultado"] == "descartar"
    assert e.estado_desde_log(sesion, "ACME") == Estado.DESCARTADA.value


def test_reemision_por_nodo_revive_descartada(sesion):
    nodo = e.crear_nodo(sesion, nombre="HBM", cluster="datacenter")
    # Entra y queda descartada por calidad pobre.
    flujo.ingresar(sesion, ticker="ACME", fecha=date(2026, 1, 1), nodo_id=nodo.id,
                   universo=_universo_ok(), filtro0=HechosFiltro0(crecimiento_yoy=0.02,
                                                                  margen_bruto=0.05))
    assert e.estado_desde_log(sesion, "ACME") == Estado.DESCARTADA.value
    # El nodo cambia → re-chequeo con hechos frescos (ahora sí califica).
    res = flujo.reemitir_por_nodo(sesion, nodo_id=nodo.id, fecha=date(2026, 2, 1),
                                  filtro0_por_ticker={"ACME": HechosFiltro0(
                                      crecimiento_yoy=0.35, margen_bruto=0.50)})
    assert res["ACME"] == "investigar"
    assert e.estado_desde_log(sesion, "ACME") == Estado.EN_DEEP_RESEARCH.value
    # No gastó Candado B (Descartada→Candidata no cuenta).
    emp = sesion.get(NxEmpresa, "ACME")
    assert emp.reentradas_usadas == 0
